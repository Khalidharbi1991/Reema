def _load_knowledge_base(self):
    """Load enhanced knowledge base with fallback to original."""
    try:
        # Try ENHANCED knowledge base first
        enhanced_path = config_manager.system_config.knowledge_dir / f"{self.name.lower()}_knowledge_ENHANCED.md"
        if enhanced_path.exists():
            with open(enhanced_path, 'r', encoding='utf-8') as f:
                self.knowledge = f.read()
            logger.info(f"{self.name}'s ENHANCED knowledge base loaded (200% expertise)", agent=self.name)
            return

        # Fallback to original
        original_path = config_manager.system_config.knowledge_dir / f"{self.name.lower()}_knowledge.md"
        if original_path.exists():
            with open(original_path, 'r', encoding='utf-8') as f:
                self.knowledge = f.read()
            logger.info(f"{self.name}'s original knowledge base loaded", agent=self.name)
    except Exception as e:
        logger.error(f"Failed to load knowledge base for {self.name}: {e}")
