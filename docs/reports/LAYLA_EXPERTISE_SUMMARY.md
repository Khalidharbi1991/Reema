```dax
MTTR Hours =
AVERAGEX(
    FILTER(Incidents, Incidents[Status] = "Closed"),
    DATEDIFF(Incidents[Created], Incidents[Resolved], HOUR)
)

SLA Compliance % =
DIVIDE(
    CALCULATE([Total Incidents], Incidents[SLAMet] = TRUE),
    [Total Incidents],

)
```
